import 'package:flutter/material.dart';
import 'package:flutter_todo/models/todo.dart';

class AddTodoDialog extends StatefulWidget {
  const AddTodoDialog({super.key});

  @override
  State<AddTodoDialog> createState() => _AddTodoDialogState();
}

class _AddTodoDialogState extends State<AddTodoDialog> {
  
  final titleController = TextEditingController();
  final descriptionController = TextEditingController();

  DateTime? dueDate;

  @override
  void dispose() {
    titleController.dispose();
    descriptionController.dispose();
    super.dispose();
  }

  void save() {
    if (titleController.text.trim().isEmpty || dueDate == null) {
      return;
    }
    Navigator.of(context).pop(
      Todo(
        title: titleController.text.trim(),
        dueDate: dueDate!,
        description: descriptionController.text.trim()
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    return AlertDialog(
      title: const Text('Add Todo'),
      content: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          TextField(
            controller: titleController,
            decoration: const InputDecoration(labelText: 'Title'),
          ),

          TextField(
            controller: descriptionController,
            decoration: const InputDecoration(labelText: 'Description'),
          ),

          const SizedBox(height: 16),

          ElevatedButton(
            onPressed: () async {
              final date = await showDatePicker(
                context: context,
                initialDate: DateTime.now(),
                firstDate: DateTime.now(),
                lastDate: DateTime(2100),
                initialEntryMode: DatePickerEntryMode.input,
              );

              if (date != null) {
                setState(() {
                  dueDate = date;
                });
              }
            },
            child: Text(
              dueDate == null
                  ? 'Select due date'
                  : '${dueDate!.day}/${dueDate!.month}/${dueDate!.year}',
            ),
          ),
        ],
      ),
      actions: [
        TextButton(
          onPressed: () => Navigator.of(context).pop(),
          child: const Text('Cancel'),
        ),
        ElevatedButton(onPressed: save, child: const Text('Add Todo')),
      ],
    );
  }
}

import 'package:flutter/material.dart';
import 'package:flutter_todo/models/todo.dart';
import 'package:flutter_todo/view/add_todo_dialog.dart';
import 'package:flutter_todo/viewmodels/todos_view_model.dart';

class TodoListItem extends StatelessWidget {
  final Todo todo;
  final TodosViewModel todosViewModel;

  const TodoListItem({
    super.key,
    required this.todo,
    required this.todosViewModel,
  });

  @override
  Widget build(BuildContext context) {
    return ListTile(
      leading: Checkbox(
        value: todo.completed,
        onChanged: (value) {
          if (value == true) {
            todosViewModel.completeTodo(todo);
          } else {
            todosViewModel.uncompleteTodo(todo);
          }
        },
      ),
      title: Text(todo.title),
      onTap: () {
        showDialog(
          context: context,
          builder: (context) => const AddTodoDialog(),
        );
      },
    );
  }
}

import 'package:flutter/material.dart';
import 'package:flutter_todo/view/add_todo_dialog.dart';
import 'package:flutter_todo/view/todo_section_dropdown.dart';
import 'package:flutter_todo/viewmodels/todos_view_model.dart';


class TodoPage extends StatefulWidget {
  const TodoPage({super.key});

  @override
  State<TodoPage> createState() => _TodoPageState();
}

class _TodoPageState extends State<TodoPage> {
  final TodosViewModel todosViewModel = TodosViewModel();

  Future<void> _addTodo() async {
    final todo = await showDialog(
      context: context,
      builder: (context) => const AddTodoDialog(),
    );

    if (todo == null) return;

    todosViewModel.addTodo(todo);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        backgroundColor: Theme.of(context).colorScheme.inversePrimary,
        title: const Text('My Todos'),
      ),
      body: AnimatedBuilder(
        animation: todosViewModel,
        builder: (context, child) {
          return ListView(
            padding: const EdgeInsets.all(16),
            children: [
              TodoSectionDropdown(
                title: 'Current',
                todos: todosViewModel.currentTodos,
                todosViewModel: todosViewModel,
              ),

              const SizedBox(height: 16),

              TodoSectionDropdown(
                title: 'Completed',
                todos: todosViewModel.completedTodos,
                todosViewModel: todosViewModel,
              ),
            ],
          );
        },
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: _addTodo,
        tooltip: 'Add Todo',
        child: const Icon(Icons.add),
      ),
    );
  }
}

import 'package:flutter/material.dart';

class TodoSectionDropdown extends StatefulWidget {

  final String title;
  final List<Todo> todos;
  final TodosViewModel todosViewModel;
  final bool openByDefault;

  const TodoSectionDropdown({
    super.key,
    required this.title,
    required this.todos,
    required this.todosViewModel,
    this.openByDefault = true
  });

  @override
  State<TodoSectionDropdown> createState() => _TodoSectionDropdownState();
}

class _TodoSectionDropdownState extends State<TodoSectionDropdown> {
  late bool expanded;

  @override
  void initState() {
    super.initState();
    expanded = widget.openByDefault;
  }

  @override
  Widget build(BuildContext context) {
    return Card(
      child: Column(
        children: [
          ListTile(
            title: Text(
              widget.title,
              style: const TextStyle(
                fontSize: 22,
                fontWeight: FontWeight.bold,
              ),
            ),
            trailing: Icon(
              expanded ? Icons.keyboard_arrow_down : Icons.keyboard_arrow_right,
            ),
            onTap: () {
              setState(() {
                expanded = !expanded;
              });
            },
          ),
          
          if(expanded)
            ...widget.todos.map(
              (todo) => TodoListItem(
                todo: todo,
                todosViewModel: widget.todosViewModel,
              ),
            ),
        ],
      )
    );
  }
}

import 'package:flutter/material.dart';
import 'package:flutter_todo/models/todo.dart';

class TodosViewModel extends ChangeNotifier {
  final List<Todo> _currentTodos = [];
  final List<Todo> _completedTodos = [];

  List<Todo> get currentTodos => List.unmodifiable(_currentTodos);
  List<Todo> get completedTodos => List.unmodifiable(_completedTodos);

  bool get isEmpty => _currentTodos.isEmpty;

  void addTodo(Todo todo)
  {
    _currentTodos.add(todo);
    notifyListeners();
  }

  void removeTodo(Todo todo) {
    _currentTodos.remove(todo);
    _completedTodos.remove(todo);
    notifyListeners();
  }

  void completeTodo(Todo todo) {
    if(_currentTodos.contains(todo)) {
      todo.completed = true;
      _completedTodos.add(todo);
      _currentTodos.remove(todo);
      notifyListeners();
    }
  }

  void uncompleteTodo(Todo todo) {
    if(_completedTodos.contains(todo)) {
      todo.completed = false;
      _currentTodos.add(todo);
      _completedTodos.remove(todo);
      notifyListeners();
    }
  }
}
